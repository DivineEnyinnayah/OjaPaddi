import { v2 as cloudinary } from "cloudinary";
cloudinary.config({
  cloud_name: process.env.CLOUDINARY_CLOUD_NAME,
  api_key: process.env.CLOUDINARY_API_KEY,
  api_secret: process.env.CLOUDINARY_API_SECRET,
});

// Generate transformed delivery URL
const transformedUrl = cloudinary.url("OjaPaddi", {
  transformation: [
    { crop: "fill", width: 1000, height: 1000, gravity: "auto" },
    { overlay: { font_family: "Arial", font_size: 72, text: "Adapt everywhere", color: "white" }, opacity: 100, width: "100%" },
 { overlay: { font_family: "Arial", font_size: 34, text: "Dynamic media built in real time", color: "rgb(233, 213, 255)" }, opacity: 100, width: "100%" },
    { shadow: 50 },
    { gravity: "south_west", x: 80, y: 140 },
    { shadow: 35 },
    { gravity: "south_west", x: 84, y: 90 },
    { format: "auto" }
  ]
});
